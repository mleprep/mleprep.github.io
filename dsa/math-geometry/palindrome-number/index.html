9. Palindrome Number

Negative numbers and numbers ending in 0 (except 0) aren't palindromes. Reverse only the second half of the digits and compare with the first half — no overflow risk.

Complexity: O(log x).

class Solution:
    def isPalindrome(self, x: int) -> bool:
        if x < 0 or (x % 10 == 0 and x != 0):
            return False
        rev = 0
        while x > rev:
            x, d = divmod(x, 10)
            rev = rev * 10 + d
        return x == rev or x == rev // 10

class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0 || (x % 10 == 0 && x != 0)) return false;
        int rev = 0;
        while (x > rev) { rev = rev * 10 + x % 10; x /= 10; }
        return x == rev || x == rev / 10;
    }
}

func isPalindrome(x int) bool {
	if x < 0 || x%10 == 0 && x != 0 {
		return false
	}
	rev := 0
	for x > rev { // reverse only the second half
		rev = rev*10 + x%10
		x /= 10
	}
	return x == rev || x == rev/10 // odd length: drop the middle digit
}
