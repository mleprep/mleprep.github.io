867. Transpose Matrix

EasyLeetCode ↗matrix
Asked at 6Bloomberg32Google29Microsoft23Meta22Amazon22Verkada76

Element (i, j) moves to (j, i), so an R × C matrix becomes C × R (a new matrix unless it's square).

Complexity: O(R·C).

class Solution {
    public int[][] transpose(int[][] matrix) {
        int R = matrix.length, C = matrix[0].length;
        int[][] t = new int[C][R];
        for (int i = 0; i < R; i++) for (int j = 0; j < C; j++) t[j][i] = matrix[i][j];
        return t;
    }
}

func transpose(matrix [][]int) [][]int {
	R, C := len(matrix), len(matrix[0])
	t := make([][]int, C)
	for c := range t {
		t[c] = make([]int, R)
		for r := 0; r < R; r++ {
			t[c][r] = matrix[r][c]
		}
	}
	return t
}
