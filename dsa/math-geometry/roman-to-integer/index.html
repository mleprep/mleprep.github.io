class Solution:
    def romanToInt(self, s: str) -> int:
        val = {"I": 1, "V": 5, "X": 10, "L": 50, "C": 100, "D": 500, "M": 1000}
        total = 0
        for i, ch in enumerate(s):
            v = val[ch]
            total += -v if i + 1 < len(s) and val[s[i + 1]] > v else v
        return total

class Solution {
    public int romanToInt(String s) {
        int total = 0;
        for (int i = 0; i < s.length(); i++) {
            int v = val(s.charAt(i));
            total += (i + 1 < s.length() && val(s.charAt(i + 1)) > v) ? -v : v;
        }
        return total;
    }

    private int val(char c) {
        switch (c) {
            case 'I': return 1; case 'V': return 5; case 'X': return 10; case 'L': return 50;
            case 'C': return 100; case 'D': return 500; default: return 1000;
        }
    }
}

func romanToInt(s string) int {
	val := map[byte]int{'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000}
	total := 0
	for i := 0; i < len(s); i++ {
		v := val[s[i]]
		if i+1 < len(s) && val[s[i+1]] > v { // a smaller symbol before a larger one subtracts
			total -= v
		} else {
			total += v
		}
	}
	return total
}
