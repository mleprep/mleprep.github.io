394. Decode String

On [ push (current string, repeat count) and reset; on ] pop and set cur = prev + cur * k.

3[a2[c]]
'3' k=3   '[' push ("",3)   'a' cur="a"
'2' k=2   '[' push ("a",2)  'c' cur="c"
']' cur = "a" + "c"*2 = "acc"
']' cur = ""  + "acc"*3 = "accaccacc"

Complexity: O(output length).

class Solution:
    def decodeString(self, s: str) -> str:
        st, cur, k = [], "", 0
        for ch in s:
            if ch.isdigit():
                k = k * 10 + int(ch)
            elif ch == "[":
                st.append((cur, k))
                cur, k = "", 0
            elif ch == "]":
                prev, times = st.pop()
                cur = prev + cur * times
            else:
                cur += ch
        return cur

assert Solution().decodeString("3[a]2[bc]") == "aaabcbc"
assert Solution().decodeString("3[a2[c]]") == "accaccacc"
assert Solution().decodeString("2[abc]3[cd]ef") == "abcabccdcdcdef"

class Solution {
    public String decodeString(String s) {
        Deque<StringBuilder> strs = new ArrayDeque<>();
        Deque<Integer> counts = new ArrayDeque<>();
        StringBuilder cur = new StringBuilder();
        int k = 0;
        for (char c : s.toCharArray()) {
            if (Character.isDigit(c)) k = k * 10 + (c - '0');
            else if (c == '[') { strs.push(cur); counts.push(k); cur = new StringBuilder(); k = 0; }
            else if (c == ']') {
                StringBuilder prev = strs.pop();
                prev.append(cur.toString().repeat(counts.pop()));
                cur = prev;
            } else cur.append(c);
        }
        return cur.toString();
    }
}

func decodeString(s string) string {
	type frame struct {
		prev  string
		times int
	}
	st, cur, k := []frame{}, "", 0
	for i := 0; i < len(s); i++ {
		ch := s[i]
		switch {
		case ch >= '0' && ch <= '9':
			k = k*10 + int(ch-'0')
		case ch == '[':
			st = append(st, frame{cur, k})
			cur, k = "", 0
		case ch == ']':
			f := st[len(st)-1]
			st = st[:len(st)-1]
			cur = f.prev + strings.Repeat(cur, f.times)
		default:
			cur += string(ch)
		}
	}
	return cur
}
