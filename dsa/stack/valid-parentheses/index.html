Problem

s contains only the brackets ()[]{}. Decide whether it is valid: every opening bracket is closed by the same type of bracket, in the correct (properly nested) order.

Example: "()[]{}" → true; "(]" → false; "([)]" → false

Push expected closers; on a closer, it must match the top.

Complexity: O(n).

class Solution:
    def isValid(self, s: str) -> bool:
        pair = {")": "(", "]": "[", "}": "{"}
        st = []
        for ch in s:
            if ch in pair:
                if not st or st.pop() != pair[ch]:
                    return False
            else:
                st.append(ch)
        return not st

assert Solution().isValid("()[]{}") is True
assert Solution().isValid("(]") is False
assert Solution().isValid("([)]") is False
assert Solution().isValid("{[]}") is True

class Solution {
    public boolean isValid(String s) {
        Deque<Character> st = new ArrayDeque<>();
        for (char c : s.toCharArray()) {
            if (c == '(') st.push(')');
            else if (c == '[') st.push(']');
            else if (c == '{') st.push('}');
            else if (st.isEmpty() || st.pop() != c) return false;
        }
        return st.isEmpty();
    }
}

func isValid(s string) bool {
	pair := map[byte]byte{')': '(', ']': '[', '}': '{'}
	st := []byte{}
	for i := 0; i < len(s); i++ {
		if open, ok := pair[s[i]]; ok {
			if len(st) == 0 || st[len(st)-1] != open {
				return false
			}
			st = st[:len(st)-1]
		} else {
			st = append(st, s[i])
		}
	}
	return len(st) == 0
}
