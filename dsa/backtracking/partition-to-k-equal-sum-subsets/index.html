698. Partition to K Equal Sum Subsets

MediumLeetCode ↗backtrackingpruning
Asked at 8Amazon31Microsoft20Google20Meta8.5Bloomberg8.3LinkedIn67Zeta64TikTok26

Target = total / k. Place numbers (sorted descending, so failures happen early) into k buckets. Prune: never put a number into a bucket that would overflow, and if placing into an empty bucket failed, other empty buckets will fail too.

Complexity: O(kⁿ) worst case, heavily pruned.

class Solution:
    def canPartitionKSubsets(self, nums: List[int], k: int) -> bool:
        total = sum(nums)
        if total % k:
            return False
        target = total // k
        nums.sort(reverse=True)
        if nums[0] > target:
            return False
        buckets = [0] * k

        def bt(i):
            if i == len(nums):
                return True
            for b in range(k):
                if buckets[b] + nums[i] <= target:
                    buckets[b] += nums[i]
                    if bt(i + 1):
                        return True
                    buckets[b] -= nums[i]
                if buckets[b] == 0:
                    break                     # symmetric empty buckets
            return False

        return bt(0)

class Solution {
    public boolean canPartitionKSubsets(int[] nums, int k) {
        int total = 0;
        for (int x : nums) total += x;
        if (total % k != 0) return false;
        Arrays.sort(nums);
        int target = total / k;
        if (nums[nums.length - 1] > target) return false;
        return bt(nums, nums.length - 1, new int[k], target);   // largest first
    }

    private boolean bt(int[] nums, int i, int[] buckets, int target) {
        if (i < 0) return true;
        for (int b = 0; b < buckets.length; b++) {
            if (buckets[b] + nums[i] <= target) {
                buckets[b] += nums[i];
                if (bt(nums, i - 1, buckets, target)) return true;
                buckets[b] -= nums[i];
            }
            if (buckets[b] == 0) break;
        }
        return false;
    }
}

func canPartitionKSubsets(nums []int, k int) bool {
	total := 0
	for _, x := range nums {
		total += x
	}
	if total%k != 0 {
		return false
	}
	target := total / k
	sort.Sort(sort.Reverse(sort.IntSlice(nums))) // big numbers first fail fast
	if nums[0] > target {
		return false
	}
	buckets := make([]int, k)
	var bt func(i int) bool
	bt = func(i int) bool {
		if i == len(nums) {
			return true
		}
		for b := 0; b < k; b++ {
			if buckets[b]+nums[i] <= target {
				buckets[b] += nums[i]
				if bt(i + 1) {
					return true
				}
				buckets[b] -= nums[i]
			}
			if buckets[b] == 0 { // empty buckets are interchangeable
				break
			}
		}
		return false
	}
	return bt(0)
}
