2709. Greatest Common Divisor Traversal

HardLeetCode ↗union-findsieve

Indices i and j are directly connected iff they share a prime factor, so connect each index to its primes: with a smallest-prime-factor sieve, factor every number and union the index with the first index seen for each of its primes. Traversal between all pairs ⇔ one connected component. A 1 (no prime factors) is isolated unless it's the only element.

Complexity: O(M log log M + n log M) for M = max(nums).

class Solution:
    def canTraverseAllPairs(self, nums: List[int]) -> bool:
        n = len(nums)
        if n == 1:
            return True
        if 1 in nums:
            return False
        M = max(nums)
        spf = list(range(M + 1))                      # smallest prime factor
        for i in range(2, int(M ** 0.5) + 1):
            if spf[i] == i:
                for j in range(i * i, M + 1, i):
                    if spf[j] == j:
                        spf[j] = i
        parent = list(range(n))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        first = {}                                    # prime -> first index having it
        for i, x in enumerate(nums):
            while x > 1:
                p = spf[x]
                if p in first:
                    parent[find(i)] = find(first[p])
                else:
                    first[p] = i
                while x % p == 0:
                    x //= p
        return len({find(i) for i in range(n)}) == 1

class Solution {
    public boolean canTraverseAllPairs(int[] nums) {
        int n = nums.length, M = 0;
        if (n == 1) return true;
        for (int x : nums) { if (x == 1) return false; M = Math.max(M, x); }
        int[] spf = new int[M + 1];
        for (int i = 2; i <= M; i++) if (spf[i] == 0) for (int j = i; j <= M; j += i) if (spf[j] == 0) spf[j] = i;
        int[] parent = new int[n];
        for (int i = 0; i < n; i++) parent[i] = i;
        Map<Integer, Integer> first = new HashMap<>();
        for (int i = 0; i < n; i++) {
            for (int x = nums[i]; x > 1; ) {
                int p = spf[x];
                Integer j = first.putIfAbsent(p, i);
                if (j != null) parent[find(parent, i)] = find(parent, j);
                while (x % p == 0) x /= p;
            }
        }
        int root = find(parent, 0);
        for (int i = 1; i < n; i++) if (find(parent, i) != root) return false;
        return true;
    }

    private int find(int[] p, int x) {
        while (p[x] != x) x = p[x] = p[p[x]];
        return x;
    }
}

func canTraverseAllPairs(nums []int) bool {
	n := len(nums)
	if n == 1 {
		return true
	}
	M := 0
	for _, x := range nums {
		if x == 1 {
			return false // 1 shares no factor with anything
		}
		M = max(M, x)
	}
	spf := make([]int, M+1) // smallest prime factor
	for i := range spf {
		spf[i] = i
	}
	for i := 2; i*i <= M; i++ {
		if spf[i] == i {
			for j := i * i; j <= M; j += i {
				if spf[j] == j {
					spf[j] = i
				}
			}
		}
	}
	parent := make([]int, n)
	for i := range parent {
		parent[i] = i
	}
	find := func(x int) int {
		for parent[x] != x {
			parent[x] = parent[parent[x]]
			x = parent[x]
		}
		return x
	}
	first := map[int]int{} // prime -> first index divisible by it
	for i, x := range nums {
		for x > 1 {
			p := spf[x]
			if j, ok := first[p]; ok {
				parent[find(i)] = find(j)
			} else {
				first[p] = i
			}
			for x%p == 0 {
				x /= p
			}
		}
	}
	for i := 1; i < n; i++ {
		if find(i) != find(0) {
			return false
		}
	}
	return true
}
