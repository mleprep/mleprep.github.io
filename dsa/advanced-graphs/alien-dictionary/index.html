269. Alien Dictionary

Problem

words are sorted lexicographically in an alien language that uses lowercase English letters in an unknown order. Return a string of the letters that appear, in an order consistent with words (any valid order), or "" if the list is inconsistent.

Example: ["wrt","wrf","er","ett","rftt"] → "wertf"; ["z","x","z"] → ""

Constraints: note ["abc","ab"] is invalid: a word can't come after its own prefix.

Compare adjacent words: the first differing letters give an edge a → b. If a word is followed by its own proper prefix, the input is invalid. Topologically sort the letters (Kahn); a cycle → "".

wrt, wrf  -> t < f
wrf, er   -> w < e
er, ett   -> r < t
ett, rftt -> e < r          order: w e r t f

Complexity: O(total characters).

class Solution:
    def alienOrder(self, words: List[str]) -> str:
        g = {ch: set() for w in words for ch in w}
        indeg = {ch: 0 for ch in g}
        for a, b in zip(words, words[1:]):
            for x, y in zip(a, b):
                if x != y:
                    if y not in g[x]:
                        g[x].add(y)
                        indeg[y] += 1
                    break
            else:
                if len(a) > len(b):
                    return ""
        q = deque(sorted(ch for ch in indeg if indeg[ch] == 0))
        out = []
        while q:
            u = q.popleft()
            out.append(u)
            for v in sorted(g[u]):
                indeg[v] -= 1
                if indeg[v] == 0:
                    q.append(v)
        return "".join(out) if len(out) == len(g) else ""

assert Solution().alienOrder(["wrt", "wrf", "er", "ett", "rftt"]) == "wertf"
assert Solution().alienOrder(["z", "x", "z"]) == ""
assert Solution().alienOrder(["abc", "ab"]) == ""

class Solution {
    public String alienOrder(String[] words) {
        Map<Character, Set<Character>> g = new TreeMap<>();
        Map<Character, Integer> indeg = new TreeMap<>();
        for (String w : words) for (char c : w.toCharArray()) { g.putIfAbsent(c, new TreeSet<>()); indeg.putIfAbsent(c, 0); }
        for (int i = 0; i + 1 < words.length; i++) {
            String a = words[i], b = words[i + 1];
            int j = 0;
            while (j < Math.min(a.length(), b.length()) && a.charAt(j) == b.charAt(j)) j++;
            if (j == Math.min(a.length(), b.length())) {
                if (a.length() > b.length()) return "";
                continue;
            }
            if (g.get(a.charAt(j)).add(b.charAt(j))) indeg.merge(b.charAt(j), 1, Integer::sum);
        }
        Deque<Character> q = new ArrayDeque<>();
        for (Map.Entry<Character, Integer> e : indeg.entrySet()) if (e.getValue() == 0) q.add(e.getKey());
        StringBuilder sb = new StringBuilder();
        while (!q.isEmpty()) {
            char u = q.poll();
            sb.append(u);
            for (char v : g.get(u)) if (indeg.merge(v, -1, Integer::sum) == 0) q.add(v);
        }
        return sb.length() == g.size() ? sb.toString() : "";
    }
}

func alienOrder(words []string) string {
	adj := map[byte]map[byte]bool{}
	indeg := map[byte]int{}
	for _, w := range words {
		for i := 0; i < len(w); i++ {
			if adj[w[i]] == nil {
				adj[w[i]] = map[byte]bool{}
				indeg[w[i]] = 0
			}
		}
	}
	for i := 0; i+1 < len(words); i++ {
		a, b := words[i], words[i+1]
		j := 0
		for j < len(a) && j < len(b) && a[j] == b[j] {
			j++
		}
		if j < len(a) && j < len(b) { // first difference: a[j] comes before b[j]
			if !adj[a[j]][b[j]] {
				adj[a[j]][b[j]] = true
				indeg[b[j]]++
			}
		} else if len(a) > len(b) {
			return "" // "abc" before "ab" is impossible
		}
	}
	q := []byte{} // Kahn's algorithm, letters taken alphabetically for a deterministic answer
	for c := byte('a'); c <= 'z'; c++ {
		if d, ok := indeg[c]; ok && d == 0 {
			q = append(q, c)
		}
	}
	out := []byte{}
	for len(q) > 0 {
		u := q[0]
		q = q[1:]
		out = append(out, u)
		for v := byte('a'); v <= 'z'; v++ {
			if adj[u][v] {
				indeg[v]--
				if indeg[v] == 0 {
					q = append(q, v)
				}
			}
		}
	}
	if len(out) < len(indeg) {
		return "" // a cycle
	}
	return string(out)
}
