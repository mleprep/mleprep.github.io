310. Minimum Height Trees

MediumLeetCode ↗topological-sortleaf-trimming
Asked at 7Google33Amazon22Bloomberg18Microsoft12Meta8.5Splunk100Stackline100

Problem

A tree has nodes 0…n − 1 and the given undirected edges. Rooting it at different nodes gives different heights. Return every root that gives the minimum height, in any order.

Example: n = 6, edges = [[3,0],[3,1],[3,2],[3,4],[5,4]] → [3,4]

Constraints: n ≤ 2·10⁴.

The best roots are the tree's centers (1 or 2 nodes). Peel leaves layer by layer (like Kahn's algorithm on degree 1) until ≤ 2 nodes remain.

0 - 1 - 2 - 3 - 4      trim leaves {0,4} -> {1,3} -> remaining {2}  = center

Complexity: O(n).

class Solution:
    def findMinHeightTrees(self, n: int, edges: List[List[int]]) -> List[int]:
        if n <= 2:
            return list(range(n))
        g = defaultdict(set)
        for a, b in edges:
            g[a].add(b)
            g[b].add(a)
        leaves = [i for i in range(n) if len(g[i]) == 1]
        remaining = n
        while remaining > 2:
            remaining -= len(leaves)
            nxt = []
            for leaf in leaves:
                nb = g[leaf].pop()
                g[nb].remove(leaf)
                if len(g[nb]) == 1:
                    nxt.append(nb)
            leaves = nxt
        return sorted(leaves)

assert sorted(Solution().findMinHeightTrees(4, [[1, 0], [1, 2], [1, 3]])) == sorted([1])
assert sorted(Solution().findMinHeightTrees(6, [[3, 0], [3, 1], [3, 2], [3, 4], [5, 4]])) == sorted([3, 4])

class Solution {
    public List<Integer> findMinHeightTrees(int n, int[][] edges) {
        if (n <= 2) { List<Integer> r = new ArrayList<>(); for (int i = 0; i < n; i++) r.add(i); return r; }
        List<Set<Integer>> g = new ArrayList<>();
        for (int i = 0; i < n; i++) g.add(new HashSet<>());
        for (int[] e : edges) { g.get(e[0]).add(e[1]); g.get(e[1]).add(e[0]); }
        List<Integer> leaves = new ArrayList<>();
        for (int i = 0; i < n; i++) if (g.get(i).size() == 1) leaves.add(i);
        int remaining = n;
        while (remaining > 2) {
            remaining -= leaves.size();
            List<Integer> nxt = new ArrayList<>();
            for (int leaf : leaves) {
                int nb = g.get(leaf).iterator().next();
                g.get(nb).remove(leaf);
                if (g.get(nb).size() == 1) nxt.add(nb);
            }
            leaves = nxt;
        }
        Collections.sort(leaves);
        return leaves;
    }
}

func findMinHeightTrees(n int, edges [][]int) []int {
	if n <= 2 {
		out := []int{}
		for i := 0; i < n; i++ {
			out = append(out, i)
		}
		return out
	}
	adj := make([][]int, n)
	deg := make([]int, n)
	for _, e := range edges {
		adj[e[0]] = append(adj[e[0]], e[1])
		adj[e[1]] = append(adj[e[1]], e[0])
		deg[e[0]]++
		deg[e[1]]++
	}
	leaves := []int{}
	for i := 0; i < n; i++ {
		if deg[i] == 1 {
			leaves = append(leaves, i)
		}
	}
	remaining := n
	for remaining > 2 { // peel leaves layer by layer; the last 1-2 nodes are the centres
		remaining -= len(leaves)
		next := []int{}
		for _, leaf := range leaves {
			for _, nb := range adj[leaf] {
				deg[nb]--
				if deg[nb] == 1 {
					next = append(next, nb)
				}
			}
		}
		leaves = next
	}
	return leaves
}
