1489. Find Critical and Pseudo-Critical Edges in Minimum Spanning Tree

HardLeetCode ↗mstunion-find
Asked at 4Amazon20Google8.2Meta5.0Uber30

Compute the MST weight once with Kruskal. Then for each edge: if the MST without it is heavier (or the graph disconnects), it's critical; otherwise, if an MST forced to include it still has the optimal weight, it's pseudo-critical. Each check is one Kruskal run.

Complexity: O(E² · α(V)) (after one sort).

class Solution:
    def findCriticalAndPseudoCriticalEdges(self, n: int, edges: List[List[int]]) -> List[List[int]]:
        order = sorted(range(len(edges)), key=lambda i: edges[i][2])

        def mst(skip=-1, force=-1):
            parent = list(range(n))

            def find(x):
                while parent[x] != x:
                    parent[x] = parent[parent[x]]
                    x = parent[x]
                return x

            weight = used = 0
            if force >= 0:
                a, b, w = edges[force]
                parent[find(a)] = find(b)
                weight, used = w, 1
            for i in order:
                a, b, w = edges[i]
                if i != skip and find(a) != find(b):
                    parent[find(a)] = find(b)
                    weight, used = weight + w, used + 1
            return weight if used == n - 1 else float("inf")

        base = mst()
        critical, pseudo = [], []
        for i in range(len(edges)):
            if mst(skip=i) > base:
                critical.append(i)
            elif mst(force=i) == base:
                pseudo.append(i)
        return [critical, pseudo]

class Solution {
    private int n;
    private int[][] edges;
    private Integer[] order;

    public List<List<Integer>> findCriticalAndPseudoCriticalEdges(int n, int[][] edges) {
        this.n = n;
        this.edges = edges;
        order = new Integer[edges.length];
        for (int i = 0; i < edges.length; i++) order[i] = i;
        Arrays.sort(order, (x, y) -> edges[x][2] - edges[y][2]);
        int base = mst(-1, -1);
        List<Integer> critical = new ArrayList<>(), pseudo = new ArrayList<>();
        for (int i = 0; i < edges.length; i++) {
            if (mst(i, -1) > base) critical.add(i);
            else if (mst(-1, i) == base) pseudo.add(i);
        }
        return List.of(critical, pseudo);
    }

    private int mst(int skip, int force) {
        int[] parent = new int[n];
        for (int i = 0; i < n; i++) parent[i] = i;
        int weight = 0, used = 0;
        if (force >= 0) { union(parent, edges[force][0], edges[force][1]); weight = edges[force][2]; used = 1; }
        for (int i : order) if (i != skip && union(parent, edges[i][0], edges[i][1])) { weight += edges[i][2]; used++; }
        return used == n - 1 ? weight : Integer.MAX_VALUE;
    }

    private boolean union(int[] p, int a, int b) {
        int ra = find(p, a), rb = find(p, b);
        if (ra == rb) return false;
        p[ra] = rb;
        return true;
    }

    private int find(int[] p, int x) {
        while (p[x] != x) x = p[x] = p[p[x]];
        return x;
    }
}

func findCriticalAndPseudoCriticalEdges(n int, edges [][]int) [][]int {
	order := make([]int, len(edges))
	for i := range order {
		order[i] = i
	}
	sort.Slice(order, func(a, b int) bool { return edges[order[a]][2] < edges[order[b]][2] })
	mst := func(skip, force int) int { // Kruskal without edge `skip` / starting with edge `force`
		parent := make([]int, n)
		for i := range parent {
			parent[i] = i
		}
		find := func(x int) int {
			for parent[x] != x {
				parent[x] = parent[parent[x]]
				x = parent[x]
			}
			return x
		}
		weight, used := 0, 0
		if force >= 0 {
			e := edges[force]
			parent[find(e[0])] = find(e[1])
			weight, used = e[2], 1
		}
		for _, i := range order {
			e := edges[i]
			if i != skip && find(e[0]) != find(e[1]) {
				parent[find(e[0])] = find(e[1])
				weight += e[2]
				used++
			}
		}
		if used < n-1 {
			return math.MaxInt // disconnected
		}
		return weight
	}
	base := mst(-1, -1)
	critical, pseudo := []int{}, []int{}
	for i := range edges {
		if mst(i, -1) > base {
			critical = append(critical, i)
		} else if mst(-1, i) == base {
			pseudo = append(pseudo, i)
		}
	}
	return [][]int{critical, pseudo}
}
