    d, u = heapq.heappop(pq)
    if d > dist.get(u, inf): continue          # stale entry
    for v, w in graph[u]:
        if d + w < dist.get(v, inf):
            dist[v] = d + w
            heapq.heappush(pq, (d + w, v))
