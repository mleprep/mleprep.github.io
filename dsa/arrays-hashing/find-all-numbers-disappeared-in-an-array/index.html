448. Find All Numbers Disappeared in an Array

EasyLeetCode ↗in-place
Asked at 6Google47Bloomberg34Microsoft33Amazon31Meta20tcs36

Problem

nums has n values, each in the range [1, n]; some appear twice and others not at all. Return every number in [1, n] that is missing.

Example: [4,3,2,7,8,2,3,1] → [5,6]

Constraints: follow-up: O(n) time and no extra space besides the output.

Values are in [1, n]: for each value v, mark index v−1 by making it negative. Indices that stay positive are the missing numbers.

Complexity: O(n) time, O(1) extra space.

class Solution:
    def findDisappearedNumbers(self, nums: List[int]) -> List[int]:
        for v in nums:
            i = abs(v) - 1
            nums[i] = -abs(nums[i])
        return [i + 1 for i, v in enumerate(nums) if v > 0]

assert Solution().findDisappearedNumbers([4, 3, 2, 7, 8, 2, 3, 1]) == [5, 6]
assert Solution().findDisappearedNumbers([1, 1]) == [2]

class Solution {
    public List<Integer> findDisappearedNumbers(int[] nums) {
        for (int v : nums) {
            int i = Math.abs(v) - 1;
            nums[i] = -Math.abs(nums[i]);
        }
        List<Integer> res = new ArrayList<>();
        for (int i = 0; i < nums.length; i++) if (nums[i] > 0) res.add(i + 1);
        return res;
    }
}

func findDisappearedNumbers(nums []int) []int {
	for _, v := range nums {
		i := abs(v) - 1
		nums[i] = -abs(nums[i]) // mark value i+1 as seen
	}
	res := []int{}
	for i, v := range nums {
		if v > 0 {
			res = append(res, i+1)
		}
	}
	return res
}

func abs(x int) int {
	if x < 0 {
		return -x
	}
	return x
}
