523. Continuous Subarray Sum

MediumLeetCode ↗prefix-sumhash-map
Asked at 10Meta60Bloomberg32Amazon30Apple30Google27Microsoft23Infosys30PhonePe53Yandex28tcs27

A subarray sum is a multiple of k ⇔ two prefix sums have the same remainder mod k. Store the first index of each remainder (seed 0 → −1); a repeat at least 2 positions later wins.

Complexity: O(n).

class Solution:
    def checkSubarraySum(self, nums: List[int], k: int) -> bool:
        first = {0: -1}
        pre = 0
        for i, x in enumerate(nums):
            pre = (pre + x) % k
            if pre in first:
                if i - first[pre] >= 2:
                    return True
            else:
                first[pre] = i
        return False

class Solution {
    public boolean checkSubarraySum(int[] nums, int k) {
        Map<Integer, Integer> first = new HashMap<>();
        first.put(0, -1);
        int pre = 0;
        for (int i = 0; i < nums.length; i++) {
            pre = (pre + nums[i]) % k;
            Integer j = first.putIfAbsent(pre, i);
            if (j != null && i - j >= 2) return true;
        }
        return false;
    }
}

func checkSubarraySum(nums []int, k int) bool {
	first := map[int]int{0: -1} // prefix sum mod k -> first index
	pre := 0
	for i, x := range nums {
		pre = (pre + x) % k
		if j, ok := first[pre]; ok {
			if i-j >= 2 {
				return true
			}
		} else {
			first[pre] = i
		}
	}
	return false
}
