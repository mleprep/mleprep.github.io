713. Subarray Product Less Than K

MediumLeetCode ↗variable-windowcounting
Asked at 21Apple68Amazon38Google34Bloomberg32Meta32Microsoft23Squarepoint Capital91Revolut77Wells Fargo65Meesho57Infosys30Samsung71Salesforce66Tekion63Airbnb57Expedia54Agoda48Walmart Labs43Oracle35tcs27Goldman Sachs21

Problem

Given positive integers nums, count the contiguous subarrays whose product is strictly less than k.

Example: nums = [10,5,2,6], k = 100 → 8

Constraints: n ≤ 3·10⁴; 1 ≤ nums[i] ≤ 1000; 0 ≤ k ≤ 10⁶.

For each right end r, shrink until product < k; then every subarray ending at r and starting in [l, r] qualifies → add r − l + 1.

Complexity: O(n) time, O(1) space.

class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        if k <= 1:
            return 0
        prod, l, count = 1, 0, 0
        for r, x in enumerate(nums):
            prod *= x
            while prod >= k:
                prod //= nums[l]
                l += 1
            count += r - l + 1
        return count

assert Solution().numSubarrayProductLessThanK([10, 5, 2, 6], 100) == 8
assert Solution().numSubarrayProductLessThanK([1, 2, 3], 0) == 0

class Solution {
    public int numSubarrayProductLessThanK(int[] nums, int k) {
        if (k <= 1) return 0;
        int prod = 1, l = 0, count = 0;
        for (int r = 0; r < nums.length; r++) {
            prod *= nums[r];
            while (prod >= k) prod /= nums[l++];
            count += r - l + 1;
        }
        return count;
    }
}

func numSubarrayProductLessThanK(nums []int, k int) int {
	if k <= 1 {
		return 0
	}
	prod, l, count := 1, 0, 0
	for r, x := range nums {
		prod *= x
		for prod >= k {
			prod /= nums[l]
			l++
		}
		count += r - l + 1 // subarrays ending at r
	}
	return count
}
