1482. Minimum Number of Days to Make m Bouquets

MediumLeetCode ↗search-on-answer
Asked at 9Bloomberg45Amazon42Google40Microsoft33Meta29Flipkart52PhonePe87Barclays53Adobe44

canMake(day) (count runs of k adjacent bloomed flowers ≥ m) is monotonic in day → binary search the day between min and max bloom day.

Complexity: O(n log max).

class Solution:
    def minDays(self, bloomDay: List[int], m: int, k: int) -> int:
        if m * k > len(bloomDay):
            return -1

        def ok(day):
            made = run = 0
            for b in bloomDay:
                run = run + 1 if b <= day else 0
                if run == k:
                    made, run = made + 1, 0
            return made >= m

        lo, hi = min(bloomDay), max(bloomDay)
        while lo < hi:
            mid = (lo + hi) // 2
            if ok(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo

assert Solution().minDays([1, 10, 3, 10, 2], 3, 1) == 3
assert Solution().minDays([1, 10, 3, 10, 2], 3, 2) == -1
assert Solution().minDays([7, 7, 7, 7, 12, 7, 7], 2, 3) == 12

class Solution {
    public int minDays(int[] bloomDay, int m, int k) {
        if ((long) m * k > bloomDay.length) return -1;
        int lo = Integer.MAX_VALUE, hi = 0;
        for (int b : bloomDay) { lo = Math.min(lo, b); hi = Math.max(hi, b); }
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            int made = 0, run = 0;
            for (int b : bloomDay) {
                run = b <= mid ? run + 1 : 0;
                if (run == k) { made++; run = 0; }
            }
            if (made >= m) hi = mid; else lo = mid + 1;
        }
        return lo;
    }
}

func minDays(bloomDay []int, m int, k int) int {
	if m*k > len(bloomDay) {
		return -1
	}
	ok := func(day int) bool {
		made, run := 0, 0
		for _, b := range bloomDay {
			if b <= day {
				run++
			} else {
				run = 0
			}
			if run == k {
				made, run = made+1, 0
			}
		}
		return made >= m
	}

		}
	}
