297. Serialize and Deserialize Binary Tree

Preorder with explicit null markers (#) uniquely identifies a tree. Deserialize by consuming tokens in the same preorder.

    1
   / \        "1,2,#,#,3,4,#,#,5,#,#"
  2   3
     / \
    4   5

Complexity: O(n) both ways.

class Codec:
    def serialize(self, root: Optional[TreeNode]) -> str:
        out = []

        def dfs(node):
            if not node:
                out.append("#")
                return
            out.append(str(node.val))
            dfs(node.left)
            dfs(node.right)

        dfs(root)
        return ",".join(out)

    def deserialize(self, data: str) -> Optional[TreeNode]:
        it = iter(data.split(","))

        def build():
            v = next(it)
            if v == "#":
                return None
            node = TreeNode(int(v))
            node.left = build()
            node.right = build()
            return node

        return build()

class Codec {
    public String serialize(TreeNode root) {
        StringBuilder sb = new StringBuilder();
        ser(root, sb);
        return sb.substring(0, sb.length() - 1);
    }

    private void ser(TreeNode n, StringBuilder sb) {
        if (n == null) { sb.append("#,"); return; }
        sb.append(n.val).append(',');
        ser(n.left, sb);
        ser(n.right, sb);
    }

    public TreeNode deserialize(String data) {
        return des(new ArrayDeque<>(Arrays.asList(data.split(","))));
    }

    private TreeNode des(Deque<String> q) {
        String v = q.poll();
        if (v.equals("#")) return null;
        TreeNode n = new TreeNode(Integer.parseInt(v));
        n.left = des(q);
        n.right = des(q);
        return n;
    }
}

type Codec struct{}

func Constructor() Codec {
	return Codec{}
}

// serialize writes the preorder with "#" for missing children.
func (this *Codec) serialize(root *TreeNode) string {
	out := []string{}
	var dfs func(*TreeNode)
	dfs = func(node *TreeNode) {
		if node == nil {
			out = append(out, "#")
			return
		}
		out = append(out, strconv.Itoa(node.Val))
		dfs(node.Left)
		dfs(node.Right)
	}
	dfs(root)
	return strings.Join(out, ",")
}

func (this *Codec) deserialize(data string) *TreeNode {
	vals, i := strings.Split(data, ","), 0
	var build func() *TreeNode
	build = func() *TreeNode {
		v := vals[i]
		i++
		if v == "#" {
			return nil
		}
		n, _ := strconv.Atoi(v)
		node := &TreeNode{Val: n}
		node.Left = build()
		node.Right = build()
		return node
	}
	return build()
}
