530. Minimum Absolute Difference in BST

EasyLeetCode ↗inorder
Asked at 3Google28Meta15Amazon10

In-order visits values in sorted order, so the minimum difference is between consecutive visited values.

Complexity: O(n).

class Solution:
    def getMinimumDifference(self, root: Optional[TreeNode]) -> int:
        best, prev = float("inf"), None

        def inorder(node):
            nonlocal best, prev
            if not node:
                return
            inorder(node.left)
            if prev is not None:
                best = min(best, node.val - prev)
            prev = node.val
            inorder(node.right)

        inorder(root)
        return best

class Solution {
    private int best = Integer.MAX_VALUE;
    private Integer prev = null;

    public int getMinimumDifference(TreeNode root) {
        inorder(root);
        return best;
    }

    private void inorder(TreeNode n) {
        if (n == null) return;
        inorder(n.left);
        if (prev != null) best = Math.min(best, n.val - prev);
        prev = n.val;
        inorder(n.right);
    }
}

func getMinimumDifference(root *TreeNode) int {
	best, prev, havePrev := math.MaxInt, 0, false
	var inorder func(*TreeNode) // in-order visits values sorted: compare neighbours
	inorder = func(node *TreeNode) {
		if node == nil {
			return
		}
		inorder(node.Left)
		if havePrev {
			best = min(best, node.Val-prev)
		}
		prev, havePrev = node.Val, true
		inorder(node.Right)
	}
	inorder(root)
	return best
}
