145. Binary Tree Postorder Traversal

EasyLeetCode ↗stackreverse-trick
Asked at 5Bloomberg14Amazon13Google13Microsoft7.5Meta5.0

Postorder (left, right, root) is the reverse of a modified preorder (root, right, left): run preorder pushing left before right, then reverse the output.

Complexity: O(n) time, O(n) space.

class Solution:
    def postorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        out, st = [], [root] if root else []
        while st:
            node = st.pop()
            out.append(node.val)
            if node.left:
                st.append(node.left)
            if node.right:
                st.append(node.right)
        return out[::-1]

class Solution {
    public List<Integer> postorderTraversal(TreeNode root) {
        LinkedList<Integer> out = new LinkedList<>();
        Deque<TreeNode> st = new ArrayDeque<>();
        if (root != null) st.push(root);
        while (!st.isEmpty()) {
            TreeNode n = st.pop();
            out.addFirst(n.val);                 // prepend = reverse of root-right-left
            if (n.left != null) st.push(n.left);
            if (n.right != null) st.push(n.right);
        }
        return out;
    }
}

func postorderTraversal(root *TreeNode) []int {
	out, st := []int{}, []*TreeNode{}
	if root != nil {
		st = append(st, root)
	}
	for len(st) > 0 { // root-right-left, then reversed = left-right-root
		node := st[len(st)-1]
		st = st[:len(st)-1]
		out = append(out, node.Val)
		if node.Left != nil {
			st = append(st, node.Left)
		}
		if node.Right != nil {
			st = append(st, node.Right)
		}
	}
	for i, j := 0, len(out)-1; i < j; i, j = i+1, j-1 {
		out[i], out[j] = out[j], out[i]
	}
	return out
}
