287. Find the Duplicate Number

MediumLeetCode ↗fast-slowcycle-detection
Asked at 18Amazon57Bloomberg50Microsoft48Google46Meta39Goldman Sachs33Nike89DP world88Citadel64Nvidia60Walmart Labs53Anduril49Flipkart43Zoho42tcs42Cisco37Salesforce28TikTok26

Values in [1, n] → view i → nums[i] as a linked list. The duplicate value is where two arrows enter the same node = cycle entrance (Floyd, like 142). No modification, O(1) space.

Complexity: O(n) time, O(1) space.

class Solution:
    def findDuplicate(self, nums: List[int]) -> int:
        slow = fast = 0
        while True:
            slow, fast = nums[slow], nums[nums[fast]]
            if slow == fast:
                break
        p = 0
        while p != slow:
            p, slow = nums[p], nums[slow]
        return p

assert Solution().findDuplicate([1, 3, 4, 2, 2]) == 2
assert Solution().findDuplicate([3, 1, 3, 4, 2]) == 3

class Solution {
    public int findDuplicate(int[] nums) {
        int slow = 0, fast = 0;
        do { slow = nums[slow]; fast = nums[nums[fast]]; } while (slow != fast);
        int p = 0;
        while (p != slow) { p = nums[p]; slow = nums[slow]; }
        return p;
    }
}

func findDuplicate(nums []int) int {
	slow, fast := 0, 0 // i -> nums[i] is a linked list whose cycle starts at the duplicate
	for {
		slow, fast = nums[slow], nums[nums[fast]]
		if slow == fast {
			break
		}
	}
	p := 0
	for p != slow {
		p, slow = nums[p], nums[slow]
	}
	return p
}
