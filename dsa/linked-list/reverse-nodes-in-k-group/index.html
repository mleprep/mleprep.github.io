25. Reverse Nodes in k-Group

HardLeetCode ↗reversaldummy-head
Asked at 24Microsoft61Amazon55Bloomberg49Google47Meta41Apple22Qualcomm100TikTok58Zopsmart57Palo Alto Networks50Cadence100Sigmoid87Arista Networks78ZScaler63Autodesk59josh technology58MathWorks56Salesforce52Cisco45Walmart Labs43Infosys39Goldman Sachs33tcs27Uber22

For each group: check k nodes exist, reverse them in place, then connect the previous group's tail to the new head and move on. A dummy head keeps the first group uniform.

k=2:  d -> 1 -> 2 -> 3 -> 4 -> 5
      d -> 2 -> 1 -> 4 -> 3 -> 5   (last group < k stays)

Complexity: O(n) time, O(1) space.

class Solution:
    def reverseKGroup(self, head: Optional[ListNode], k: int) -> Optional[ListNode]:
        dummy = ListNode(0, head)
        group_prev = dummy
        while True:
            kth = group_prev
            for _ in range(k):
                kth = kth.next
                if not kth:
                    return dummy.next
            group_next = kth.next
            prev, cur = group_next, group_prev.next      # reverse [group_prev.next .. kth]
            while cur is not group_next:
                cur.next, prev, cur = prev, cur, cur.next
            first = group_prev.next
            group_prev.next = kth
            group_prev = first

assert list_vals(Solution().reverseKGroup(make_list([1, 2, 3, 4, 5]), 2)) == [2, 1, 4, 3, 5]
assert list_vals(Solution().reverseKGroup(make_list([1, 2, 3, 4, 5]), 3)) == [3, 2, 1, 4, 5]

class Solution {
    public ListNode reverseKGroup(ListNode head, int k) {
        ListNode dummy = new ListNode(0, head), groupPrev = dummy;
        while (true) {
            ListNode kth = groupPrev;
            for (int i = 0; i < k; i++) {
                kth = kth.next;
                if (kth == null) return dummy.next;
            }
            ListNode groupNext = kth.next, prev = groupNext, cur = groupPrev.next;
            while (cur != groupNext) { ListNode nxt = cur.next; cur.next = prev; prev = cur; cur = nxt; }
            ListNode first = groupPrev.next;
            groupPrev.next = kth;
            groupPrev = first;
        }
    }
}

func reverseKGroup(head *ListNode, k int) *ListNode {
	dummy := &ListNode{Next: head}
	groupPrev := dummy
	for {
		kth := groupPrev
		for i := 0; i < k; i++ {
			kth = kth.Next
			if kth == nil { // fewer than k nodes left: keep them as they are
				return dummy.Next
			}
		}
		groupNext := kth.Next
		prev, cur := groupNext, groupPrev.Next // reverse [groupPrev.Next .. kth]
		for cur != groupNext {
			cur.Next, prev, cur = prev, cur, cur.Next
		}
		first := groupPrev.Next
		groupPrev.Next = kth
		groupPrev = first
	}
}
