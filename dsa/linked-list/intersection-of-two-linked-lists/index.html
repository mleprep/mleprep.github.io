160. Intersection of Two Linked Lists

EasyLeetCode ↗two-pointers
Asked at 12Amazon47Microsoft46Bloomberg45Google38Meta34Apple30TikTok26Airbnb75Wells Fargo65Samsung63LinkedIn47Goldman Sachs28

Walk a over A then B, b over B then A. Both travel lenA + lenB steps, so they meet at the intersection (or both reach null together).

Complexity: O(m + n) time, O(1) space.

class Solution:
    def getIntersectionNode(self, headA: ListNode, headB: ListNode) -> Optional[ListNode]:
        a, b = headA, headB
        while a is not b:
            a = a.next if a else headB
            b = b.next if b else headA
        return a

class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        ListNode a = headA, b = headB;
        while (a != b) {
            a = a == null ? headB : a.next;
            b = b == null ? headA : b.next;
        }
        return a;
    }
}

func getIntersectionNode(headA, headB *ListNode) *ListNode {
	a, b := headA, headB
	for a != b { // both walk lenA + lenB nodes, so they meet at the junction (or at nil)
		if a != nil {
			a = a.Next
		} else {
			a = headB
		}
		if b != nil {
			b = b.Next
		} else {
			b = headA
		}
	}
	return a
}
