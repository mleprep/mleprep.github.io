72. Edit Distance

MediumLeetCode ↗two-strings

dp[i][j] = edits to turn a[:i] into b[:j]. Match → dp[i−1][j−1]; else 1 + min(replace dp[i−1][j−1], delete dp[i−1][j], insert dp[i][j−1]). Base: dp[i][0] = i, dp[0][j] = j.

class Solution:
    def minDistance(self, word1: str, word2: str) -> int:
        n = len(word2)
        dp = list(range(n + 1))
        for i, a in enumerate(word1, 1):
            diag, dp[0] = dp[0], i
            for j in range(1, n + 1):
                up = dp[j]
                dp[j] = diag if a == word2[j - 1] else 1 + min(diag, up, dp[j - 1])
                diag = up
        return dp[n]

assert Solution().minDistance("horse", "ros") == 3
assert Solution().minDistance("intention", "execution") == 5

class Solution {
    public int minDistance(String a, String b) {
        int m = a.length(), n = b.length();
        int[][] dp = new int[m + 1][n + 1];
        for (int i = 0; i <= m; i++) dp[i][0] = i;
        for (int j = 0; j <= n; j++) dp[0][j] = j;
        for (int i = 1; i <= m; i++)
            for (int j = 1; j <= n; j++)
                dp[i][j] = a.charAt(i - 1) == b.charAt(j - 1) ? dp[i - 1][j - 1]
                        : 1 + Math.min(dp[i - 1][j - 1], Math.min(dp[i - 1][j], dp[i][j - 1]));
        return dp[m][n];
    }
}

func minDistance(word1 string, word2 string) int {
	n := len(word2)
	dp := make([]int, n+1) // dp[j] = edits turning word1[:i] into word2[:j]
	for j := range dp {
		dp[j] = j
	}
	for i := 1; i <= len(word1); i++ {
		diag := dp[0]
		dp[0] = i
		for j := 1; j <= n; j++ {
			up := dp[j]
			if word1[i-1] == word2[j-1] {
				dp[j] = diag
			} else {
				dp[j] = 1 + min(diag, up, dp[j-1]) // replace, delete, insert
			}
			diag = up
		}
	}
	return dp[n]
}
