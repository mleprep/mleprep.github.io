96. Unique Binary Search Trees

MediumLeetCode ↗catalan
Asked at 10Meta38Amazon31Microsoft20Google20Bloomberg18Snap80Clari79Swiggy69TikTok26LinkedIn20

Pick root r: left subtree uses r−1 nodes, right uses n−r. G(n) = Σ G(r−1)·G(n−r) — the Catalan numbers.

Complexity: O(n²).

class Solution:
    def numTrees(self, n: int) -> int:
        G = [1] + [0] * n
        for m in range(1, n + 1):
            G[m] = sum(G[r - 1] * G[m - r] for r in range(1, m + 1))
        return G[n]

assert Solution().numTrees(3) == 5
assert Solution().numTrees(1) == 1

class Solution {
    public int numTrees(int n) {
        long[] G = new long[n + 1];
        G[0] = 1;
        for (int m = 1; m <= n; m++)
            for (int r = 1; r <= m; r++) G[m] += G[r - 1] * G[m - r];
        return (int) G[n];
    }
}

func numTrees(n int) int {
	G := make([]int, n+1) // G[m] = BSTs on m keys (Catalan numbers)
	G[0] = 1
	for m := 1; m <= n; m++ {
		for r := 1; r <= m; r++ { // root r: r-1 keys on the left, m-r on the right
			G[m] += G[r-1] * G[m-r]
		}
	}
	return G[n]
}
