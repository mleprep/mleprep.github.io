368. Largest Divisible Subset

MediumLeetCode ↗lis
Asked at 7Google38Bloomberg30Meta29Microsoft25Amazon24Tekion63Oracle27

Sort. If a | b and b | c then a | c, so it's LIS with "divides" instead of "<". Keep parent pointers to rebuild the subset.

Complexity: O(n²).

class Solution:
    def largestDivisibleSubset(self, nums: List[int]) -> List[int]:
        nums.sort()
        n = len(nums)
        size, parent = [1] * n, [-1] * n
        for i in range(n):
            for j in range(i):
                if nums[i] % nums[j] == 0 and size[j] + 1 > size[i]:
                    size[i], parent[i] = size[j] + 1, j
        i = max(range(n), key=size.__getitem__)
        res = []
        while i != -1:
            res.append(nums[i])
            i = parent[i]
        return res[::-1]

assert Solution().largestDivisibleSubset([1, 2, 3]) in ([1, 2], [1, 3])
assert Solution().largestDivisibleSubset([1, 2, 4, 8]) == [1, 2, 4, 8]

class Solution {
    public List<Integer> largestDivisibleSubset(int[] nums) {
        Arrays.sort(nums);
        int n = nums.length, best = 0;
        int[] size = new int[n], parent = new int[n];
        for (int i = 0; i < n; i++) {
            size[i] = 1; parent[i] = -1;
            for (int j = 0; j < i; j++)
                if (nums[i] % nums[j] == 0 && size[j] + 1 > size[i]) { size[i] = size[j] + 1; parent[i] = j; }
            if (size[i] > size[best]) best = i;
        }
        LinkedList<Integer> res = new LinkedList<>();
        for (int i = best; i != -1; i = parent[i]) res.addFirst(nums[i]);
        return res;
    }
}

func largestDivisibleSubset(nums []int) []int {
	sort.Ints(nums)
	n := len(nums)
	size, parent := make([]int, n), make([]int, n) // chain length ending at i, previous index
	best := 0
	for i := 0; i < n; i++ {
		size[i], parent[i] = 1, -1
		for j := 0; j < i; j++ {
			if nums[i]%nums[j] == 0 && size[j]+1 > size[i] {
				size[i], parent[i] = size[j]+1, j
			}
		}
		if size[i] > size[best] {
			best = i
		}
	}
	res := []int{}
	for i := best; i != -1; i = parent[i] {
		res = append([]int{nums[i]}, res...)
	}
	return res
}
