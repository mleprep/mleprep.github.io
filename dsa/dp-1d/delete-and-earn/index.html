740. Delete and Earn

MediumLeetCode ↗take-or-skipreduction
Asked at 12Amazon28Microsoft28Google17Bloomberg14Meta12Walmart Labs82IBM44Akuna Capital97Morgan Stanley86Salesforce57TikTok35Infosys30

Taking value v earns v · count[v] and forbids v−1 and v+1 → exactly House Robber over the value axis.

Complexity: O(n + max).

class Solution:
    def deleteAndEarn(self, nums: List[int]) -> int:
        gain = [0] * (max(nums) + 1)
        for x in nums:
            gain[x] += x
        prev = cur = 0
        for g in gain:
            prev, cur = cur, max(cur, prev + g)
        return cur

assert Solution().deleteAndEarn([3, 4, 2]) == 6
assert Solution().deleteAndEarn([2, 2, 3, 3, 3, 4]) == 9

class Solution {
    public int deleteAndEarn(int[] nums) {
        int mx = 0;
        for (int x : nums) mx = Math.max(mx, x);
        int[] gain = new int[mx + 1];
        for (int x : nums) gain[x] += x;
        int prev = 0, cur = 0;
        for (int g : gain) { int nxt = Math.max(cur, prev + g); prev = cur; cur = nxt; }
        return cur;
    }
}

func deleteAndEarn(nums []int) int {
	gain := make([]int, slices.Max(nums)+1) // taking value v earns v * count(v)
	for _, x := range nums {
		gain[x] += x
	}
	prev, cur := 0, 0 // then it is House Robber over the values
	for _, g := range gain {
		prev, cur = cur, max(cur, prev+g)
	}
	return cur
}
