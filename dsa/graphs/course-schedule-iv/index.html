1462. Course Schedule IV

MediumLeetCode ↗transitive-closurefloyd-warshall
Asked at 5Amazon20Meta12Google11Uber50TikTok46

Problem

prerequisites[i] = [a, b] means course a must be taken before course b, and prerequisites are transitive. For each query [u, v], answer whether u is a (direct or indirect) prerequisite of v.

Example: numCourses = 2, prerequisites = [[1,0]], queries = [[0,1],[1,0]] → [false,true]

Constraints: up to 100 courses and 10⁴ queries.

Many reachability queries on a small graph (n ≤ 100) → precompute the transitive closure once: Floyd–Warshall on booleans (reach[i][j] |= reach[i][k] and reach[k][j]), then answer each query in O(1). (A BFS from every node or a topological pass with bitsets also works.)

Complexity: O(n³ + q).

class Solution:
    def checkIfPrerequisite(self, numCourses: int, prerequisites: List[List[int]], queries: List[List[int]]) -> List[bool]:
        n = numCourses
        reach = [[False] * n for _ in range(n)]
        for a, b in prerequisites:
            reach[a][b] = True
        for k in range(n):
            for i in range(n):
                if reach[i][k]:
                    row_k, row_i = reach[k], reach[i]
                    for j in range(n):
                        if row_k[j]:
                            row_i[j] = True
        return [reach[u][v] for u, v in queries]

assert Solution().checkIfPrerequisite(2, [[1, 0]], [[0, 1], [1, 0]]) == [False, True]
assert Solution().checkIfPrerequisite(2, [], [[1, 0], [0, 1]]) == [False, False]
assert Solution().checkIfPrerequisite(3, [[1, 2], [1, 0], [2, 0]], [[1, 0], [1, 2]]) == [True, True]
assert Solution().checkIfPrerequisite(4, [[0, 1], [1, 2], [2, 3]], [[0, 3], [3, 0], [1, 3]]) == [True, False, True]

class Solution {
    public List<Boolean> checkIfPrerequisite(int numCourses, int[][] prerequisites, int[][] queries) {
        int n = numCourses;
        boolean[][] reach = new boolean[n][n];
        for (int[] p : prerequisites) reach[p[0]][p[1]] = true;
        for (int k = 0; k < n; k++)
            for (int i = 0; i < n; i++)
                if (reach[i][k]) for (int j = 0; j < n; j++) if (reach[k][j]) reach[i][j] = true;
        List<Boolean> res = new ArrayList<>();
        for (int[] q : queries) res.add(reach[q[0]][q[1]]);
        return res;
    }
}

func checkIfPrerequisite(numCourses int, prerequisites [][]int, queries [][]int) []bool {
	n := numCourses
	reach := make([][]bool, n)
	for i := range reach {
		reach[i] = make([]bool, n)
	}
	for _, p := range prerequisites {
		reach[p[0]][p[1]] = true
	}
	for k := 0; k < n; k++ { // Floyd-Warshall transitive closure
		for i := 0; i < n; i++ {
			if reach[i][k] {
				for j := 0; j < n; j++ {
					if reach[k][j] {
						reach[i][j] = true
					}
				}
			}
		}
	}
	res := make([]bool, len(queries))
	for i, q := range queries {
		res[i] = reach[q[0]][q[1]]
	}
	return res
}
