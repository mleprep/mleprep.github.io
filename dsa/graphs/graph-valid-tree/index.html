261. Graph Valid Tree

MediumLeetCode ↗★ Blind 75union-find🔒 Premium
Asked at 7Google28Meta28Microsoft12Zenefits100Snowflake48LinkedIn43Uber22

Problem

Decide whether the undirected edges over nodes 0…n − 1 form a valid tree: connected, with no cycles.

Example: n = 5, edges = [[0,1],[0,2],[0,3],[1,4]] → true; n = 5, edges = [[0,1],[1,2],[2,3],[1,3],[1,4]] → false

A tree on n nodes has exactly n − 1 edges and no cycle (equivalently: connected). Union-Find: any union of already-connected nodes → cycle.

Complexity: O(E · α(V)).

class Solution:
    def validTree(self, n: int, edges: List[List[int]]) -> bool:
        if len(edges) != n - 1:
            return False
        parent = list(range(n))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        for a, b in edges:
            ra, rb = find(a), find(b)
            if ra == rb:
                return False
            parent[ra] = rb
        return True

assert Solution().validTree(5, [[0, 1], [0, 2], [0, 3], [1, 4]]) is True
assert Solution().validTree(5, [[0, 1], [1, 2], [2, 3], [1, 3], [1, 4]]) is False

class Solution {
    public boolean validTree(int n, int[][] edges) {
        if (edges.length != n - 1) return false;
        int[] parent = new int[n];
        for (int i = 0; i < n; i++) parent[i] = i;
        for (int[] e : edges) {
            int ra = find(parent, e[0]), rb = find(parent, e[1]);
            if (ra == rb) return false;
            parent[ra] = rb;
        }
        return true;
    }

    private int find(int[] p, int x) {
        while (p[x] != x) { p[x] = p[p[x]]; x = p[x]; }
        return x;
    }
}

func validTree(n int, edges [][]int) bool {
	if len(edges) != n-1 { // a tree has exactly n-1 edges
		return false
	}
	parent := make([]int, n)
	for i := range parent {
		parent[i] = i
	}
	find := func(x int) int {
		for parent[x] != x {
			parent[x] = parent[parent[x]]
			x = parent[x]
		}
		return x
	}
	for _, e := range edges {
		ra, rb := find(e[0]), find(e[1])
		if ra == rb { // ...and no cycle
			return false
		}
		parent[ra] = rb
	}
	return true
}
