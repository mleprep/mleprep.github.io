130. Surrounded Regions

MediumLeetCode ↗border-dfs
Asked at 12Amazon46Bloomberg44Google40Microsoft35Meta33PornHub87Nutanix60Anduril49Adobe44Oracle35TikTok26Uber22

An O survives only if connected to the border. Mark border-connected Os as safe (S) via DFS, then flip remaining O → X and S → O.

Complexity: O(R·C).

class Solution:
    def solve(self, board: List[List[str]]) -> None:
        R, C = len(board), len(board[0])

        def mark(r, c):
            if 0 <= r < R and 0 <= c < C and board[r][c] == "O":
                board[r][c] = "S"
                mark(r + 1, c); mark(r - 1, c); mark(r, c + 1); mark(r, c - 1)

        for r in range(R):
            mark(r, 0); mark(r, C - 1)
        for c in range(C):
            mark(0, c); mark(R - 1, c)
        for r in range(R):
            for c in range(C):
                board[r][c] = "O" if board[r][c] == "S" else "X"

class Solution {
    public void solve(char[][] board) {
        int R = board.length, C = board[0].length;
        for (int r = 0; r < R; r++) { mark(board, r, 0); mark(board, r, C - 1); }
        for (int c = 0; c < C; c++) { mark(board, 0, c); mark(board, R - 1, c); }
        for (char[] row : board)
            for (int c = 0; c < C; c++) row[c] = row[c] == 'S' ? 'O' : 'X';
    }

    private void mark(char[][] b, int r, int c) {
        if (r < 0 || c < 0 || r >= b.length || c >= b[0].length || b[r][c] != 'O') return;
        b[r][c] = 'S';
        mark(b, r + 1, c); mark(b, r - 1, c); mark(b, r, c + 1); mark(b, r, c - 1);
    }
}

func solve(board [][]byte) {
	R, C := len(board), len(board[0])
	var mark func(r, c int) // "O"s connected to the border are safe
	mark = func(r, c int) {
		if r >= 0 && r < R && c >= 0 && c < C && board[r][c] == 'O' {
			board[r][c] = 'S'
			mark(r+1, c)
			mark(r-1, c)
			mark(r, c+1)
			mark(r, c-1)
		}
	}
	for r := 0; r < R; r++ {
		mark(r, 0)
		mark(r, C-1)
	}
	for c := 0; c < C; c++ {
		mark(0, c)
		mark(R-1, c)
	}
	for r := 0; r < R; r++ {
		for c := 0; c < C; c++ {
			if board[r][c] == 'S' {
				board[r][c] = 'O'
			} else {
				board[r][c] = 'X'
			}
		}
	}
}
